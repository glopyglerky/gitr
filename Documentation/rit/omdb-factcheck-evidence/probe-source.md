## src/lib.rs

```rust
#![no_std]

use omdb::{Database, Error, TableDefinition, TransactionError};

pub const REPO_METADATA: TableDefinition<&str, &[u8]> = TableDefinition::new("repo_metadata");

#[derive(Debug, PartialEq, Eq)]
pub enum WriteOutcome {
    Committed,
    Busy,
}

/// Try once. The caller decides whether and when to retry a busy writer.
pub fn try_put(db: &Database, key: &str, value: &[u8]) -> Result<WriteOutcome, Error> {
    let write = match db.try_begin_write() {
        Ok(write) => write,
        Err(TransactionError::WouldBlock) => return Ok(WriteOutcome::Busy),
        Err(error) => return Err(error.into()),
    };
    {
        let mut table = write.open_table(REPO_METADATA)?;
        table.insert(key, value)?;
    }
    write.commit()?;
    Ok(WriteOutcome::Committed)
}
```

## src/main.rs

```rust
use omdb::backends::InMemoryBackend;
use omdb::{Database, Error, ReadableDatabase, ReadableTable};
use omdb_api_probe::{REPO_METADATA, WriteOutcome, try_put};

fn main() -> Result<(), Error> {
    let db = Database::builder()
        .set_cache_size(1024 * 1024)
        .create_with_backend(InMemoryBackend::new())?;
    let held = db.begin_write()?;
    assert_eq!(try_put(&db, "repo", b"blocked")?, WriteOutcome::Busy);
    drop(held);
    assert_eq!(try_put(&db, "repo", b"first")?, WriteOutcome::Committed);
    let first = ReadableDatabase::begin_read(&db)?;
    assert_eq!(try_put(&db, "repo", b"second")?, WriteOutcome::Committed);
    let old_table = first.open_table(REPO_METADATA)?;
    assert_eq!(ReadableTable::get(&old_table, "repo")?.unwrap().value(), b"first");
    let current = ReadableDatabase::begin_read(&db)?;
    let current_table = current.open_table(REPO_METADATA)?;
    assert_eq!(ReadableTable::get(&current_table, "repo")?.unwrap().value(), b"second");
    println!("PASS: built-in memory backend, WouldBlock while writer held, success after release, committed bytes, and reader snapshot isolation");
    Ok(())
}
```

## src/bin/pasted.rs

```rust
use std::sync::Mutex;
use omdb::{BackendError, StorageBackend};

pub struct MemoryStorageBackend {
    storage: Mutex<Vec<u8>>,
}

impl MemoryStorageBackend {
    pub fn new() -> Self {
        Self {
            storage: Mutex::new(Vec::new()),
        }
    }
}

impl StorageBackend for MemoryStorageBackend {
    fn len(&self) -> Result<u64, BackendError> {
        Ok(self.storage.lock().unwrap().len() as u64)
    }

    fn set_len(&self, len: u64) -> Result<(), BackendError> {
        let mut guard = self.storage.lock().unwrap();
        guard.resize(len as usize, 0);
        Ok(())
    }

    fn sync_data(&self, _eventual: bool) -> Result<(), BackendError> {
        Ok(())
    }

    fn read(&self, offset: u64, len: usize) -> Result<Box<[u8]>, BackendError> {
        let guard = self.storage.lock().unwrap();
        let start = offset as usize;
        let end = start + len;
        if end > guard.len() {
            return Err(BackendError::Other("Read out of bounds"));
        }
        Ok(guard[start..end].to_vec().into_boxed_slice())
    }

    fn write(&self, offset: u64, data: &[u8]) -> Result<(), BackendError> {
        let mut guard = self.storage.lock().unwrap();
        let start = offset as usize;
        let end = start + data.len();
        if end > guard.len() {
            guard.resize(end, 0);
        }
        guard[start..end].copy_from_slice(data);
        Ok(())
    }
}

fn main() {
    let _db = omdb::Database::builder()
        .create_with_backend(MemoryStorageBackend::new())
        .unwrap();
}
```
